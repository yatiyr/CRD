# CEIR-18q/18a-1: the repo root, so the .ckir emitter/identity tests reach assets/compute/scene_light_cull.ckir (a

# CEIR-35a Q1: GENERATE a manifest of EVERY committed .ckir so the asset-load SWEEP gate (test_ckir_asset.cpp) proves
# ckir_read accepts all of them — the honest close of the CEIR-35b over-rejection claim (the ~20 hand-picked [asset]
# gates cover only a third of the 40+ committed assets; the rest are loaded only by engine/other-exe paths). A generated
# `const char*[]` header keeps crd-kir-tests' LINK surface unchanged (⛔ ADR-0098 isolation smoke — no crd-platform link
# for directory enumeration). CONFIGURE_DEPENDS re-globs on the next build so a NEW asset joins the sweep with zero edit.
file(GLOB _crd_ckir_assets CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets/ckir/*.ckir")
set(_crd_ckir_list "")
foreach(_f ${_crd_ckir_assets})
    get_filename_component(_n "${_f}" NAME)
    string(APPEND _crd_ckir_list "    \"${_n}\",\n")
endforeach()
list(LENGTH _crd_ckir_assets _crd_ckir_count)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gen/ckir_asset_list.hpp"
    "// GENERATED by tests/gpu/kir/CMakeLists.txt (CEIR-35a Q1 asset sweep) - do not edit; re-globs on reconfigure.\n"
    "#pragma once\n"
    "namespace crd::kir::test {\n"
    "inline constexpr const char* kCommittedCkir[] = {\n${_crd_ckir_list}};\n"
    "inline constexpr int kCommittedCkirCount = ${_crd_ckir_count};\n"
    "} // namespace crd::kir::test\n")
target_include_directories(crd-kir-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gen")

crd_enable_default_pch(crd-kir-tests)

catch_discover_tests(crd-kir-tests)
